    <script>
        let db = null;
        const tableName = "250k release type";

        // Tier-1 Manufacturers Priority list (Ranked higher silently without visual badges)
        const tierOneManufacturers = [
            "Sun Pharmaceutical Industries Ltd", "Intas Pharmaceuticals Ltd", "Cipla Ltd",
            "Torrent Pharmaceuticals Ltd", "Alkem Laboratories Ltd", "Lupin Ltd",
            "Zydus Cadila", "Micro Labs Ltd", "Abbott", "Macleods Pharmaceuticals Pvt Ltd",
            "Mankind Pharma Ltd", "Dr Reddy's Laboratories Ltd", "Glenmark Pharmaceuticals Ltd",
            "Alembic Pharmaceuticals Ltd", "Ipca Laboratories Ltd", "Ajanta Pharma Ltd"
        ];

        async function initDatabase() {
            const statusEl = document.getElementById('status');
            try {
                const sqlPromise = initSqlJs({
                    locateFile: file => `https://cdnjs.cloudflare.com/ajax/libs/sql.js/1.8.0/${file}`
                });
                // Updated to point directly to your renamed database file
                const dataPromise = fetch('250k med.db').then(res => {
                    if (!res.ok) throw new Error("Database file '250k med.db' not found.");
                    return res.arrayBuffer();
                });

                const [SQL, buf] = await Promise.all([sqlPromise, dataPromise]);
                db = new SQL.Database(new Uint8Array(buf));
                
                statusEl.className = "text-xs px-3 py-1 bg-emerald-900 text-emerald-300 font-bold rounded-full";
                statusEl.innerText = "✓ Engine Ready (240k+ SKUs Loaded)";
                
                const inputEl = document.getElementById('searchInput');
                inputEl.disabled = false;
                
                if (inputEl.value.trim().length >= 1) {
                    handleSearch();
                }
            } catch (err) {
                statusEl.className = "text-xs px-3 py-1 bg-rose-900 text-rose-300 font-bold rounded-full";
                statusEl.innerText = "Error: " + err.message;
            }
        }

        function handleSearch() {
            if (!db) return;
            const query = document.getElementById('searchInput').value.trim().toLowerCase();
            const resultsTable = document.getElementById('resultsTable');
            const countEl = document.getElementById('resultsCount');

            if (!query || query.length === 0) {
                resultsTable.innerHTML = `<tr><td colspan="6" class="p-10 text-center text-slate-400">Enter brand name or API to search...</td></tr>`;
                countEl.innerText = "Waiting for search input...";
                return;
            }

            try {
                const stmt = db.prepare(`
                    SELECT * FROM "${tableName}" 
                    WHERE "Brand Name" LIKE :q OR "API (Active Ingredients Only)" LIKE :q 
                    LIMIT 100
                `);
                
                stmt.bind({ ':q': `%${query}%` });
                let rows = [];
                while (stmt.step()) {
                    rows.push(stmt.getAsObject());
                }
                stmt.free();

                if (rows.length === 0) {
                    resultsTable.innerHTML = `<tr><td colspan="6" class="p-10 text-center text-slate-400">No matching medicines found. Try searching by active ingredient.</td></tr>`;
                    countEl.innerText = "0 results found";
                    return;
                }

                // Sort results placing Tier-1 manufacturers first silently
                rows.sort((a, b) => {
                    let aTier = tierOneManufacturers.includes(a['Manufacturer']) ? 0 : 1;
                    let bTier = tierOneManufacturers.includes(b['Manufacturer']) ? 0 : 1;
                    return aTier - bTier;
                });

                countEl.innerText = `Displaying top ${rows.length} matches`;

                resultsTable.innerHTML = rows.map(r => {
                    const brand = r['Brand Name'] || '-';
                    const mfr = r['Manufacturer'] || '-';
                    const api = r['API (Active Ingredients Only)'] || '-';
                    const strength = r['Exact Strength Group'] || '-';
                    const pack = r['Pack Size'] || '-';

                    return `
                        <tr onclick="openSubstituteModal('${escapeAttr(api)}', '${escapeAttr(strength)}', '${escapeAttr(brand)}')" 
                            class="hover:bg-teal-50/60 cursor-pointer transition border-b border-slate-100">
                            <td class="p-3.5 font-bold text-teal-900">${escapeHtml(brand)}</td>
                            <td class="p-3.5 text-slate-600">${escapeHtml(mfr)}</td>
                            <td class="p-3.5 text-slate-800 font-medium">${escapeHtml(api)}</td>
                            <td class="p-3.5 text-slate-700">${escapeHtml(strength)}</td>
                            <td class="p-3.5 text-slate-500">${escapeHtml(pack)}</td>
                            <td class="p-3.5 text-center">
                                <button class="px-2.5 py-1 bg-teal-100 text-teal-800 hover:bg-teal-200 rounded-lg text-xs font-bold">
                                    Substitutes
                                </button>
                            </td>
                        </tr>
                    `;
                }).join('');
            } catch (e) {
                console.error(e);
                resultsTable.innerHTML = `<tr><td colspan="6" class="p-10 text-center text-rose-500">Query error: ${e.message}</td></tr>`;
            }
        }

        function openSubstituteModal(api, strength, brandName) {
            document.getElementById('modalBrand').innerText = brandName;
            document.getElementById('modalComposition').innerText = `API: ${api} | Strength: ${strength}`;

            const listEl = document.getElementById('substitutesList');
            listEl.innerHTML = `<tr><td colspan="3" class="p-6 text-center text-slate-400">Loading exact substitutes...</td></tr>`;
            document.getElementById('substituteModal').classList.remove('hidden');

            try {
                const stmt = db.prepare(`
                    SELECT * FROM "${tableName}" 
                    WHERE "API (Active Ingredients Only)" = :api 
                    AND "Exact Strength Group" = :strength
                `);
                
                stmt.bind({ ':api': api, ':strength': strength });
                let subs = [];
                while (stmt.step()) {
                    subs.push(stmt.getAsObject());
                }
                stmt.free();

                // Sort substitutes placing Tier-1 manufacturers on top silently
                subs.sort((a, b) => {
                    let aTier = tierOneManufacturers.includes(a['Manufacturer']) ? 0 : 1;
                    let bTier = tierOneManufacturers.includes(b['Manufacturer']) ? 0 : 1;
                    return aTier - bTier;
                });

                if (subs.length === 0) {
                    listEl.innerHTML = `<tr><td colspan="3" class="p-6 text-center text-slate-400">No alternative brands found for this exact formulation.</td></tr>`;
                } else {
                    listEl.innerHTML = subs.map(s => `
                        <tr class="hover:bg-slate-50 border-b border-slate-100">
                            <td class="p-3 font-bold text-slate-800">${escapeHtml(s['Brand Name'])}</td>
                            <td class="p-3 text-slate-600">${escapeHtml(s['Manufacturer'])}</td>
                            <td class="p-3 text-slate-500">${escapeHtml(s['Pack Size'])}</td>
                        </tr>
                    `).join('');
                }
            } catch (e) {
                console.error(e);
                listEl.innerHTML = `<tr><td colspan="3" class="p-6 text-center text-rose-500">Error loading substitutes.</td></tr>`;
            }
        }

        function closeModal() {
            document.getElementById('substituteModal').classList.add('hidden');
        }

        function escapeHtml(str) {
            return String(str).replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;').replace(/"/g, '&quot;');
        }

        function escapeAttr(str) {
            return String(str).replace(/'/g, "\\'").replace(/"/g, '&quot;');
        }

        window.onload = initDatabase;
    </script>
